### # 30: Foreign Agent Caught > reveals one hostile pact
### =============================================
hms_counterintel.30 = {
    type = country_event
    placement = root

    title = hms_counterintel.30.t
    desc = hms_counterintel.30.d
    flavor = hms_counterintel.30.f

    event_image = {
        video = "gfx/event_pictures/ip3_policeraid.bk2"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_skull.dds"

    duration = 1

    trigger = {

        has_variable = hms_counterintel_sweep_active
        NOT = { has_variable = hms_ci_agent_caught_fired }
        
        # Must actually have someone targeting us
        OR = {

            any_country = {
                has_diplomatic_pact = { who = prev type = arm_radicals }
            }
            any_country = {
                has_diplomatic_pact = { who = prev type = steal_tech }
            }
            any_country = {
                has_diplomatic_pact = { who = prev type = encourage_piracy }
            }
            any_country = {
                has_diplomatic_pact = { who = prev type = support_slave_abolitionist_movement }
            }

            # ADD HERE ANY FUTURE HOSTILE ACTIONS THAT COULD BE REVEALED
        }
    }

    immediate = {
        set_variable = hms_ci_agent_caught_fired

        # Find the hostile actor > prioritize arm_radicals, then steal_tech, then piracy, then slave movement
        if = {
            limit = {
                any_country = {
                    has_diplomatic_pact = { who = prev type = arm_radicals }
                }
            }
            random_country = {
                limit = {
                    has_diplomatic_pact = { who = prev type = arm_radicals }
                }
                save_scope_as = hms_caught_spy_country
            }
            set_variable = hms_caught_spy_type_radicals
        }
        else_if = {
            limit = {
                any_country = {
                    has_diplomatic_pact = { who = prev type = steal_tech }
                }
            }
            random_country = {
                limit = {
                    has_diplomatic_pact = { who = prev type = steal_tech }
                }
                save_scope_as = hms_caught_spy_country
            }
            set_variable = hms_caught_spy_type_tech
        }
        else_if = {
            limit = {
                any_country = {
                    has_diplomatic_pact = { who = prev type = encourage_piracy }
                }
            }
            random_country = {
                limit = {
                    has_diplomatic_pact = { who = prev type = encourage_piracy }
                }
                save_scope_as = hms_caught_spy_country
            }
            set_variable = hms_caught_spy_type_piracy
        }
        else_if = {
            limit = {
                any_country = {
                    has_diplomatic_pact = { who = prev type = encourage_piracy }
                }
            }
            random_country = {
                limit = {
                    has_diplomatic_pact = { who = prev type = support_slave_abolitionist_movement }
                }
                save_scope_as = hms_caught_spy_country
            }
            set_variable = hms_caugth_spy_slave_revolt
        }
    }

    # Option A: Expel the agent and confront the country publicly
    option = {
        name = hms_counterintel.30.a

        # Reveal the hostile country => relations hit
        change_relations = {
            country = scope:hms_caught_spy_country
            value = -25
        }

        # Remove the hostile pact
        if = {
            limit = { has_variable = hms_caught_spy_type_radicals }
            scope:hms_caught_spy_country = {
                remove_diplomatic_pact = { country = root type = arm_radicals }
                set_variable = {
                    name = caught_by
                    value = prev
                }
                trigger_event = { id = reaction_events.1 }
            }
        }
        else_if = {
            limit = { has_variable = hms_caught_spy_type_tech }
            scope:hms_caught_spy_country = {
                remove_diplomatic_pact = { country = root type = steal_tech }
                set_variable = {
                    name = caught_by
                    value = prev
                }
                trigger_event = { id = reaction_events.1 }
            }
        }
        else_if = {
            limit = { has_variable = hms_caught_spy_type_piracy }
            scope:hms_caught_spy_country = {
                remove_diplomatic_pact = { country = root type = encourage_piracy }
                set_variable = {
                    name = caught_by
                    value = prev
                }
                trigger_event = { id = reaction_events.1 }
            }
        }
        else_if = {
            limit = { has_variable = hms_caugth_spy_slave_revolt }
            scope:hms_caught_spy_country = {
                remove_diplomatic_pact = { country = root type = support_slave_abolitionist_movement }
                set_variable = {
                    name = caught_by
                    value = prev
                }
                trigger_event = { id = reaction_events.1 }
            }
        }

        # Cleanup type flags
        remove_variable = hms_caught_spy_type_radicals
        remove_variable = hms_caught_spy_type_tech
        remove_variable = hms_caught_spy_type_piracy
        remove_variable = hms_caugth_spy_slave_revolt

        ai_chance = { base = 70 }

        default_option = yes
    }

    # Option B: Keep the agent under surveillance and do not alarm the rival country
    option = {
        name = hms_counterintel.30.b

        # Cleanup type flags
        remove_variable = hms_caught_spy_type_radicals
        remove_variable = hms_caught_spy_type_tech
        remove_variable = hms_caught_spy_type_piracy
        remove_variable = hms_caugth_spy_slave_revolt

        ai_chance = { base = 30 }
    }
}
