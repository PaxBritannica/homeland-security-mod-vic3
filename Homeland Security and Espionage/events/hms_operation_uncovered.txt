namespace = hms_operation_uncovered

# INITIATOR RECEIVES NOTICE HIS OPERATION HAS BEEN DISCOVERED
hms_operation_uncovered.1 = {
    type = country_event

    title = hms_operation_uncovered.1.t
    desc = hms_operation_uncovered.1.d
    flavor = hms_operation_uncovered.1.f

    event_image = {
        video = "southamerica_aristocrats"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

    icon = "gfx/interface/icons/event_icons/event_protest.dds"

    duration = 2

    immediate = {
		if = {
			limit = {
				any_country = {
					root = {
						has_diplomatic_pact = {
							who = prev
							type = siphon_funds
						}
					}
				}
			}
			random_country = {
				limit = {
					root = {
						has_diplomatic_pact = {
							who = prev
							type = siphon_funds
						}
					}
				}
				save_scope_as = operation_target
			}
		}
		else = {
			random_country = {
				limit = {
					root = {
						has_diplomatic_pact = {
							who = prev
							type = siphon_funds
						}
					}
				}
				save_scope_as = operation_target
			}
		}
		root = {
			save_scope_as = operation_initiator
		}
	}

    option = {
        name = hms_operation_uncovered.1.a
        default_option = yes


        custom_tooltip = {
            text = we_will_stop_operation_tt
        }
        
        add_modifier = {
            name = modifier_failed_intelligence_operation 
            days = 365
        }

        change_relations = {
            country = scope:operation_target
            value = -45
        }

        remove_diplomatic_pact = {
            country = scope:operation_target
            type = siphon_funds
        }

        scope:operation_target = {
            trigger_event = {
                id = hms_operation_uncovered.2
                popup = yes
            }
        }
    
    }
}
