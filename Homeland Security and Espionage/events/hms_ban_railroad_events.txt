ban_railroad.2 = { # FIRES AT 50% RAILROAD
	type = country_event
	placement = root

	title = ban_railroad.2.t
	desc = ban_railroad.2.d
    flavor = ban_railroad.2.f

	icon = "gfx/interface/icons/event_icons/event_industry.dds"

    event_image = {
		video = "unspecific_trains"
	}

	trigger = {
		has_variable = ban_railroad_construction_active
		NOT = { has_variable = ban_railroad_completed }
		NOT = { has_variable = reached_middle_milestone_railroad }
	}

	immediate = {
		set_variable = reached_middle_milestone_railroad
	}

	duration = 3

	# Option A: Stay the course
	option = {
		name = ban_railroad.2.a

		# First batch of railroads spawned
		ban_spawn_railroads_batch = yes

		trigger_event = {
			id = ban_railroad.3
			months = 8
		}

		ai_chance = { base = 70 }
	}

	# Option B: Invest more to accelerate
	option = {
		name = ban_railroad.2.b

		ban_spawn_railroads_batch = yes
		
        add_modifier = {
            name = railways_faster_build_modifier
            months = 4
        }

		custom_tooltip = {
			text = ban_tensions_among_railroad_workers_tt
			trigger_event = {
				id = ban_railroad.3
				months = 5
			}
		}

		ai_chance = { base = 30 }
	}
}

# A disease has struck our camps
ban_railroad.13 = {
	type = country_event
	placement = root

	title = ban_railroad.13.t
	desc = ban_railroad.13.d
	flavor = ban_railroad.13.f

	event_image = {
		video = "unspecific_sick_in_hospital"
	}

	duration = 1

	trigger = {
		has_variable = ban_railroad_construction_active
		NOT = { has_variable = ban_railroad_completed }
	}

	cooldown = {
		years = 2
	}

	immediate = {
		set_variable = ban_rr_disease_fired
	}

	# Option A: Fund medical aid and quarantine — halt work temporarily
	option = {
		name = ban_railroad.13.a

		add_modifier = {
			name = unexpected_cost_railroad
			months = 7
		}

		custom_tooltip = { # slightly negative
			text = negative_progress_railroad_tt
			set_variable = {
				name = var_healing_workers
				months = 9
			}
		}

		add_technology_progress = {
			technology = pharmaceuticals
			progress = 2000
		}

		# Workers appreciate the care
		scope:ban_target = {
			add_loyalists = {
				value = 0.02
			}

			add_technology_progress = {
				technology = pharmaceuticals
				progress = 2000
			}
		}

		ai_chance = { base = 60 }
	}

	# Option B: Continue construction, replace the fallen
	option = {
		name = ban_railroad.13.b

		add_modifier = {
			name = ban_more_mortality_infrastructure_modifier
			months = 13
		}

		custom_tooltip = { # completely left workers to die
			text = negative_progress_railroad_tt
			set_variable = {
				name = var_refused_curing_workers
				months = 12
			}
		}

		scope:ban_target = {
			add_radicals = {
				value = 0.05
			}

			add_modifier = {
				name = ban_more_mortality_infrastructure_modifier
				months = 13
			}
		}

		add_radicals = {
			value = 0.03
		}

		ai_chance = { base = 40 }
	}
}
