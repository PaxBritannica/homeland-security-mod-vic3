ban_survey.2 = { # FIRES AT 50% PROSPECTING
	type = country_event
	placement = root

	title = ban_survey.2.t
	desc = ban_survey.2.d
	flavor = ban_survey.2.f

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	event_image = {
		video = "europenorthamerica_gold_prospectors"
	}

	trigger = {
		has_variable = ban_prospecting_project_active
		NOT = { has_variable = ban_prospecting_project_completed }
		NOT = { has_variable = reached_middle_milestone_prospecting }
	}

	immediate = {
		set_variable = reached_middle_milestone_prospecting
	}

	duration = 3

	# Share findings with target
	option = {
		name = ban_survey.2.a

		change_relations = {
			country = scope:ban_target
			value = 10
		}

		custom_tooltip = {
			text = ban_gain_prospecting_progress_small_tt
			je:hms_je_build_nation ?= {
				add_progress = { value = 5 name = prospecting_initiative_progress_bar }
			}
		}

		ai_chance = { base = 40 }
	}

	# Option B: Keep findings classified AND use as trade leverage
	option = {
		name = ban_survey.2.b

		change_relations = {
			country = scope:ban_target
			value = -5
		}

		# Useless for now but we could use later to influence trust bar 
		set_variable = {
			name = avoided_sharing_initial_findings
			months = 13
		}

		trigger_event = {
			id = develop_nation_events.7
			months = 6
		}

		ai_chance = { base = 30 }
	}

	# Option C: Expand the survey, our geologists are learning too
	option = {
		name = ban_survey.2.c

		add_modifier = {
			name = unexpected_cost_prospecting
			months = 7
		}

		# Funder gets domestic mining boost — expertise gained abroad
		add_modifier = {
			name = permanent_boost_to_prospecting_expertise
			months = -1
		}

		ai_chance = { base = 30 }
	}
}

# Event 4: Survey Complete (~12 months in)
ban_survey.4 = {
	type = country_event
	placement = root

	title = ban_survey.4.t
	desc = ban_survey.4.d

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

    event_image = {
        video = "unspecific_signed_contract"
    }

	trigger = {
		has_variable = ban_prospecting_project_active
		NOT = { has_variable = ban_prospecting_project_completed }
	}

	duration = 3

	option = {
		name = ban_survey.4.a

		default_option = yes

		# Remove in-progress modifiers
		remove_modifier = ban_prospecting_funding_cost

		scope:ban_target = {
			remove_modifier = ban_prospecting_in_progress
		}

		### Re-add prospecting bonus for a year to give target country more chance to discover minerals
		scope:ban_target = {
			add_modifier = {
				name = ban_prospecting_in_progress
				years = 2
			}
		}

		# # Mark as done
		set_variable = ban_prospecting_project_completed
	    remove_variable = ban_prospecting_project_active

		#final handshake
		change_relations = {
			country = scope:ban_target
			value = 20
		}

		ai_chance = { base = 100 }
	}
}

	}

		has_variable = ban_prospecting_project_active
		NOT = { has_variable = ban_prospecting_project_completed }
	}

	cooldown = {
		years = 2
	}

	immediate = {
		
		# Pick a random country with interest on target
		random_country = {
			limit = {

				NOT = { THIS = ROOT }

			}

			}

		add_modifier = {
			name = settled_land_dispute_modifier
			months = 13
		}

		ai_chance = { base = 50 }
	}

		name = ban_survey.13.b

		scope:rival_claim_country = {
			change_relations = {
				country = root
				value = -15
			}

		ai_chance = { base = 50 }
	}
