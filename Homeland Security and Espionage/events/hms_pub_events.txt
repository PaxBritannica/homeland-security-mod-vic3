scientific_pub_events.32 = {
    type = country_event
    placement = root

    title = scientific_pub_events.32.t
    desc = scientific_pub_events.32.d
    flavor = scientific_pub_events.32.f

    event_image = {
        video = "gfx/event_pictures/unspecific_signed_contract.bk2"
    }

    on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
    icon = "gfx/interface/icons/event_icons/event_default.dds"

    duration = 1

    trigger = {
        NOT = { has_variable = hms_pub_praise_fired }

        # At least one non-partner country exists
        any_country = {
            NOT = { this = root }
            NOT = { this = root.var:hms_pub_partner }
            country_rank >= rank_value:minor_power
            relations:root >= relations_threshold:cold
        }
    }

    immediate = {
        set_variable = hms_pub_praise_fired

        # get Partner ()
        root.var:hms_pub_partner = {
            save_scope_as = pub_partner
        }

        # --- HELPER COUNTRY 1 ---
        # Try major power, cold+ relations first
        if = {
            limit = {
                any_country = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    country_rank >= rank_value:major_power
                    relations:root >= relations_threshold:cold
                }
            }
            random_country = {
                limit = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    country_rank >= rank_value:major_power
                    relations:root >= relations_threshold:cold
                }
                root = {
                    set_variable = { name = interested_country_1 value = prev years = 1 }
                }
                save_scope_as = helper_country_1
            }
        }
        # Fallback: any minor power+
        else = {
            random_country = {
                limit = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    country_rank >= rank_value:minor_power
                }
                root = {
                    set_variable = { name = interested_country_1 value = this years = 1 }
                }
                save_scope_as = helper_country_1
            }
        }

        # --- HELPER COUNTRY 2 ---
        # Try major power, cold+ relations, different from helper 1
        if = {
            limit = {
                any_country = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    NOT = { this = scope:helper_country_1 }
                    country_rank >= rank_value:major_power
                    relations:root >= relations_threshold:cold
                }
            }
            random_country = {
                limit = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    NOT = { this = scope:helper_country_1 }
                    country_rank >= rank_value:major_power
                    relations:root >= relations_threshold:cold
                }
                root = {
                    set_variable = { name = interested_country_2 value = prev years = 1 }
                }
                save_scope_as = helper_country_2
            }
        }
        # Fallback: any minor power+, different from helper 1
        else_if = {
            limit = {
                any_country = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    NOT = { this = scope:helper_country_1 }
                    country_rank >= rank_value:minor_power
                }
            }
            random_country = {
                limit = {
                    NOT = { this = root }
                    NOT = { this = root.var:hms_pub_partner }
                    NOT = { this = scope:helper_country_1 }
                    country_rank >= rank_value:minor_power
                }
                root = {
                    set_variable = { name = interested_country_2 value = this years = 1 }
                }
                save_scope_as = helper_country_2
            }
        }
        # If no second country found, helper_country_2 won't exist — options handle this
    }

    # Option A: Invite reviewers to contribute progress
    option = {
        name = scientific_pub_events.32.a

        custom_tooltip = {
            text = minor_progress_tt
            add_progress_effect_small = yes
        }

        scope:helper_country_1 = {
            change_relations = {
                country = prev
                value = 10
            }
        }

        scope:helper_country_2 = {
            change_relations = {
                country = prev
                value = 10
            }
        }

        ai_chance = { base = 50 }
    }

    # Option B: Thank them and press on
    option = {
        name = scientific_pub_events.32.b
        default_option = yes

        random_list = {
            60 = {
                custom_tooltip = {
                    text = rival_countries_publication_tt

                    set_variable = { name = set_started_rival_publication years = 1 }
                    
                    set_variable = {
                        name = rival_publisher_1
                        value = scope:helper_country_1
                        years = 1
                    }

                    set_variable = {
                        name = rival_publisher_2
                        value = scope:helper_country_2
                        years = 1
                    }

                }
            }
            40 = {
                custom_tooltip = {
                    text = minor_setback_tt
                    add_setback_effect_small = yes
                }
            }
        }

        change_relations = {
            country = scope:pub_partner
            value = 20
        }

        ai_chance = { base = 50 }
    }
}
