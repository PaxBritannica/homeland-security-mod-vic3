	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 1

	trigger = {
        has_journal_entry = hms_je_sabotage
	}

    immediate = {
        # Pick first friendly helper
        random_country = {
			limit = {

                NOT = { THIS = ROOT }
                NOT = { is_country_type = decentralized }
                has_strategic_adjacency = root
                relations:root >= relations_threshold:cold
			}
			save_scope_as = helper_country_1
		}

        # Pick second friendly helper, excluding the first
        random_country = {
			limit = {
                
                NOT = { THIS = ROOT }
                NOT = { is_country_type = decentralized }
                NOT = { THIS = scope:helper_country_1}
                has_strategic_adjacency = root
                relations:root >= relations_threshold:cold
			}
			save_scope_as = helper_country_2
		}
    }

   

    option = { # Accept help from first country
        name = sabotage_event.5.a

        trigger = {
            exists = scope:helper_country_1
        }

        root = {
            change_relations = {
                country = scope:helper_country_1
                value = 25
            }

            set_variable = {
                name = accepted_foreign_assistance
                months = 12
            }
        }

        if = {
            limit = {
                has_journal_entry = hms_je_sabotage
            }
            je:hms_je_sabotage ?= {
                add_progress = { value = 2 name = sabotage_police_progress_bar }
            }

    }

    option = { # Accept help from second country
        name = sabotage_event.5.b

        trigger = {

        root = {

                value = 25

                name = accepted_foreign_assistance
                months = 12
            }

            limit = {

            }
