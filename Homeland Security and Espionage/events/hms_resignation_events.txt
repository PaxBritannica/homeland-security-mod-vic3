# A Favor for a friend...
hms_resignation_events.2 = {
    type = country_event
    placement = root
    
    title = hms_resignation_events.2.t
    desc = hms_resignation_events.2.d
    flavor = hms_resignation_events.2.f

	gui_window = event_window_1char_tabloid
	left_icon = scope:resigning_politician
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    
    trigger = {
         # Must have Secret Police law active
		 has_law = law_type:law_secret_police
		 
		 # Must have at least one viable interest group with a leader
		 any_interest_group = {
			 is_marginal = no
			 leader = {
				 is_busy = no
				#  NOT = { is_ruler = yes } # Optional: avoid targeting the ruler
			 }
		 }
    }

	cooldown = { days = stupidly_long_modifier_time }
    
    immediate = {
        # Save the randomly selected leader to a scope
        random_interest_group = {
            limit = {
                is_marginal = no
                leader = {
                    is_busy = no
                }
            }
            save_scope_as = resignation_ig
            
            leader = {
                save_scope_as = resigning_politician
            }
        }
    }
    
    option = {
		name = hms_resignation_events.2.a
		default_option = yes
	
		add_modifier = {
			name = modifier_accepted_secret_service_bribe
			days = 365
		}

	}

	option = {
		name = hms_resignation_events.2.b
		default_option = yes
	
		add_modifier = {
			name = unhappy_state_apparatus
			days = 365
		}

	}

}

# This Leader is in government 
hms_resignation_events.3 = {
    type = country_event
    placement = root
    
    title = hms_resignation_events.3.t
    desc = hms_resignation_events.3.d
    flavor = hms_resignation_events.3.f

	# Journals tipped off ?

	gui_window = event_window_1char_tabloid
	left_icon = scope:resigning_politician

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
    
    trigger = {
         # Must have Secret Police law active
		 has_law = law_type:law_secret_police
		 
		 # Must have at least one viable interest group with a leader
		 any_interest_group = {
			 is_marginal = no
			 leader = {
				 is_busy = no
				#  NOT = { is_ruler = yes } # Optional: avoid targeting the ruler
			 }
		 }
    }

	cooldown = { days = long_modifier_time }
    
    immediate = {
        # Save the randomly selected leader to a scope
        random_interest_group = {
            limit = {
                is_marginal = no
                leader = {
                    is_busy = no
                }
            }
            save_scope_as = resignation_ig
            
            leader = {
                save_scope_as = resigning_politician
            }
        }

		random_country = {
			limit = {
				country_rank >= rank_value:great_power
				NOT = {
					this = root
				}
				has_technology_researched = HMS_tech_era_2_secret_service
			}
			save_scope_as = foreign_power
		}


    }
    
    option = {
		name = hms_resignation_events.3.a
		default_option = yes
	
	}

	option = {
		name = hms_resignation_events.3.b 

		scope:resigning_politician = {

			add_modifier = {
				name = shamed_as_foreign_spy
			}

		}

		random_list = {
			75 = {}
			25 = {
				scope:resigning_politician = {
					kill_character = yes
				}
			}
		}

		root = {
			change_relations = {
				country = scope:foreign_power
				value = -25
			}
		}

	}

	option = {
		name = hms_resignation_events.3.c

		trigger = {
			NOT = { has_law_or_variant = law_type:law_secret_police }
		}

	}

	option = {
		name = hms_resignation_events.3.e

		trigger = {
			has_law_or_variant = law_type:law_secret_police
		}

		# Secret Service will work him out of the office quietly

		custom_tooltip = {
			text = secret_service_handles_tt

				kill_character = yes
			}
		}
