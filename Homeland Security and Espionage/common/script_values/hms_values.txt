piracy_naval_modifier = {
    value = 1
    
    # If initiator has more navy (ratio > 1), add up to +50%
    if = {
        limit = {
            piracy_naval_difference > 1
        }
        # Add bonus, but cap at +0.5 (so max total is 1.5)
        if = {
            limit = {
                piracy_naval_difference > 1.5
            }
            add = 0.5
        }
        else = {
            add = piracy_naval_difference
            subtract = 1
        }
    }
    
    # If target has more navy (ratio < 1), reduce down to 0.5x minimum
    else_if = {
        limit = {
            piracy_naval_difference < 1
        }
        multiply = piracy_naval_difference
        # Cap at 0.5x minimum
        if = {
            limit = {
                piracy_naval_difference < 0.5
            }
            divide = piracy_naval_difference
            multiply = 0.5
        }
    }
}

formation_naval_awareness = {

	every_combat_unit = {
		add = {
			add = flotilla_naval_awareness_value
			multiply = this.manpower
		}
	}

	if = {
		limit = {
			any_combat_unit = {
				count >= 1
				this.manpower >= 1
			}
		}
		divide = {
			every_combat_unit = {
				add = this.manpower
			}
		}
	}

	else = {
		value = 0
	}
}
