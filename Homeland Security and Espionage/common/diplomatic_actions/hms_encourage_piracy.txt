encourage_piracy = {
	texture = "gfx/interface/icons/lens_toolbar_icons/encourage_piracy.dds"

    groups = {
        general
    }

    requires_approval = no

    should_notify_third_parties = no

    unlocking_technologies = {
        admiralty
    }

	potential = {
		NOT = { is_country_type = decentralized }
		scope:target_country = {
			NOT = { is_country_type = decentralized }
		}
	}
	
	possible = {
		is_revolutionary = no
		in_default = no
	
		NOT = {
			OR = {

				# NO ALLIANCE, DEFENSIVE PACT OR MUTUAL NON INTERFERENCE TREATY ALLOWED FOR THIS ACTION
				# WE ARE MEN WITH VALUES, NOT DOUBLE-DEALING DIPLOMATS
				
				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = mutual_non_interference
					}
				}

				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = defensive_pact
					}
				}

				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = alliance
					}
				}
	
				has_war_with = scope:target_country
			}
		}

        # INVESTMENT LEVEL NEEDED IN FOREIGN AFFAIRS
        ROOT = {
            # country_rank >= rank_value:minor_power Every nation can yarrr
            has_law_or_variant = law_type:law_office_of_foreign_affairs
            institution_investment_level = {
                institution = institution_foreign_affairs
                value >= 3
            }
        }

		custom_tooltip = {
			text = has_neither_bribed_or_fought_pirates_tt

			
			NOR = {
					scope:target_country = {
						has_variable = recently_fought_pirates
						
					}
			
					scope:target_country = {
						has_variable = recently_bribed_pirates 
						
					}
					
					scope:target_country = {
						has_truce_with = root
					}
			}
		}

		custom_tooltip = {
			text = target_has_not_researched_sea_lanes_strategy_tt

			scope:target_country = {
				NOT = { has_technology_researched = sea_lanes_strategy }
			}
		}
	
		has_diplomatic_relevance = scope:target_country

        can_reach_target_country = yes #needs to have port or sea route

    }

    pact = {
        cost = 200
	
		is_two_sided_pact = yes
		show_in_outliner = no
		
		income_transfer = 0.07
		max_paying_country_income_to_transfer = 1
		second_country_gets_income_transfer = yes
		income_transfer_based_on_second_country = yes

		income_transfer_to_pops = {
			allow_non_fully_accepted = yes
			upper_strata_pops = {
				value = 0
				max = 1
				min = 0
			}
			middle_strata_pops = {
				value = 0.2
				max = 1
				min = 0				
			}
			lower_strata_pops = {
				value = 0.8
				max = 1
				min = 0				
			}
		}

        # we might wanna move this to on_actions later on 

        monthly_effect = {
			custom_tooltip = {
				text = encourage_piracy_monthly_effect_tt
				random_list = { 
					16 = {} # balance ? 
					1 = { 
						root = {
							trigger_event = {
					            id = hostile_diplo_events.4
					        }
					    }
					}
				}
			}
		}

		should_invalidate = {
			OR = {		
				is_country_type = decentralized

				scope:target_country = { is_country_type = decentralized }

				in_default = yes

				scope:target_country = {
					has_war_with = root								
				}

				scope:target_country = { has_variable = won_against_pirates }
				scope:target_country = { has_variable = recently_bribed_pirates }
			}
		}
    }

    ai = {

		propose_score = {
			value = 10
			if = {
				limit = {
					scope:target_country = {
						has_diplomatic_pact = {
							who = root
							type = rivalry
						}
					}
				}
			}
		}
	
		will_propose = { 
			scaled_debt <= 0
			relations:root <= relations_threshold:cold
            can_reach_target_country = yes
		}
		
		will_break = { 
			OR = {		
				scaled_debt > 0.25
				AND = {
					scaled_debt > 0
					is_at_war = yes
				}
				NOT = {
					scope:target_country = {
						relations:root <= relations_threshold:cold								
					}

				}						
			}					
		}			
	}

}
