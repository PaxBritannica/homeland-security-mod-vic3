support_slave_abolitionist_movement = {
	texture = "gfx/interface/icons/lens_toolbar_icons/support_slave_abolitionist_movement.dds"

	groups = {
		general
	}

	requires_approval = no
	should_notify_third_parties = no

	unlocking_technologies = {
		egalitarianism
	}
	
	potential = {
		NOT = { is_country_type = decentralized }

		scope:target_country = {
			NOT = { is_country_type = decentralized }
			NOT = { has_law = law_type:law_slavery_banned }
		}
	}
	
	possible = {

        # INVESTMENT LEVEL NEEDED IN FOREIGN AFFAIRS
        ROOT = {
            country_rank >= rank_value:major_power

            has_law_or_variant = law_type:law_office_of_foreign_affairs
            institution_investment_level = {
                institution = institution_foreign_affairs
                value >= 3
            }
			
			has_law_or_variant = law_type:law_slavery_banned
        }

		#HAS DIPLOMATIC INTEREST
		has_diplomatic_relevance = scope:target_country

		NOT = {
			OR = {

                # NO ALLIANCE, DEFENSIVE PACT OR MUTUAL NON INTERFERENCE TREATY ALLOWED FOR THIS ACTION
				# WE ARE MEN WITH VALUES, NOT DOUBLE-DEALING DIPLOMATS

                any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = mutual_non_interference
					}
				}

				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = defensive_pact
					}
				}

				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = alliance
					}
				}
	
				has_war_with = scope:target_country
			}
		}
	}
	
	pact = {
		cost = 150
	
		is_two_sided_pact = no
		show_in_outliner = yes
		can_be_used_in_sway_offers = no
		sway_maneuvers_cost = 10
		
		income_transfer = 0.05
		max_paying_country_income_to_transfer = 1
		second_country_gets_income_transfer = yes
		income_transfer_based_on_second_country = yes

		income_transfer_to_pops = {
			allow_non_fully_accepted = yes
			upper_strata_pops = {
				value = 0
				max = 1
				min = 0
			}
			middle_strata_pops = {
				value = 0
				max = 1
				min = 0				
			}
			lower_strata_pops = {
				value = 1
				max = 1
				min = 0				
			}
		}

		first_modifier = {
			building_intel_hq_throughput_mult = -0.1
		}

		second_modifier = {
			# Finding slaves will be harder
			# Render slave commerce inefficient and empower local slaves
			state_slave_import_mult = -0.5
			country_slaves_pol_str_mult = 0.5
			building_slaves_job_attractiveness_mult = -0.3
			building_employment_slaves_mult = -0.3
		}

		monthly_effect = {
			custom_tooltip = {
				text = support_slaves_movement_tt
				random_list = { 
					35 = {}
					1 = { 
						root = {
							trigger_event = {
					            id = hostile_diplo_events.3
					        }
					    }
					}
				}								
			}
		} 	
	
		is_about_to_auto_break = {
            scope:target_country.relations:root <= -17 # this: poor + 3
        }

		should_auto_break = {
			custom_tooltip = {
				text = DIPLO_ACTION_RELATIONS_POOR_OR_LESS

                # make sure to update is_about_to_auto_break when changing this:
				scope:target_country.relations:root <= relations_threshold:poor
			}
		}
	
		should_invalidate = {
			OR = {
				has_war_with = scope:target_country				
				is_country_type = decentralized
				scope:target_country = { is_country_type = decentralized }
				scope:target_country = { has_law = law_type:law_slavery_banned } #Invalidate once slaves free
				in_default = yes
			}
		}
	}
	
	ai = {

		propose_score = {
			value = 10
		}
	
		will_propose = { 
			scaled_debt <= 0
			is_at_war = no
			scope:target_country = {
				has_diplomatic_pact = {
					who = root
					type = rivalry
				}
				relations:root <= relations_threshold:cold									
			}
		}
		
		will_break = { 
			OR = {		
				scaled_debt > 0.25
				AND = {
					scaled_debt > 0
					is_at_war = yes
				}
				scope:target_country = {
					has_diplomatic_pact = {
						has_law = law_type:law_slavery_banned
					}										
				}
				NOT = {
					scope:target_country = {
						has_diplomatic_pact = {
							who = root
							type = rivalry
						}										
					}
				}						
			}					
		}			
	}
}
