steal_tech = {
    texture = "gfx/interface/icons/lens_toolbar_icons/steal_tech.dds"

    groups = {
        general
    }
    
    requires_approval = no
    should_notify_third_parties = no

    selectable = {
        NOT = { is_country_type = decentralized }
        is_revolutionary = no
    }

    potential = {
        scope:target_country = {
            NOT = { is_country_type = decentralized }
        }

        institution_investment_level = {
            institution = institution_foreign_affairs
            value >= 3 # balance ? Higher investment level ?
        }

        # No need we can steal from everybody as long as it has more techs than us
        # scope:target_country = {
		# 	country_rank > rank_value:minor_power
		# }

        # has_game_rule = enable_tech_stealing
    }

    possible = {
        is_revolutionary = no
        is_default = no
        NOT = {
            has_war_with = scope:target_country
        }

        has_law_or_variant = law_type:law_office_of_foreign_affairs

        # TARGET COUNTRY MUST HAVE MORE TECH THAN US
        scope:target_country.techs_researched > root.techs_researched
        has_diplomatic_relevance = scope:target_country

        NOT = {
			OR = {

                # NO ALLIANCE, DEFENSIVE PACT OR MUTUAL NON INTERFERENCE TREATY ALLOWED FOR THIS ACTION
				# WE ARE MEN WITH VALUES, NOT DOUBLE-DEALING DIPLOMATS

                any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = mutual_non_interference
					}
				}

				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = defensive_pact
					}
				}

				any_scope_treaty = {
					binds = root
					binds = scope:target_country
					any_scope_article = {
						has_type = alliance
					}
				}
	
				has_war_with = scope:target_country
			}

            scope:target_country =  {
                has_variable = successful_tech_defense_against_thieves
            }
		}
    }

    pact = {
        cost = 150

        is_two_sided_pact = yes
        show_in_outliner = no
        can_be_used_in_sway_offers = no

        income_transfer = 0.7
		max_paying_country_income_to_transfer = 1
		second_country_gets_income_transfer = yes
		income_transfer_based_on_second_country = yes

		income_transfer_to_pops = {
			allow_non_fully_accepted = no
			upper_strata_pops = {
				value = 0.5
				max = 1
				min = 0
			}
			middle_strata_pops = {
				value = 0.3
				max = 1
				min = 0				
			}
			lower_strata_pops = {
				value = 0.2
				max = 1
				min = 0				
			}
		}
        
        first_modifier = {
			building_intel_hq_throughput_mult = -0.1
        }

        second_modifier = {}
        
        monthly_effect = {
            custom_tooltip = {
                text = steal_technology_monthly_effect_tt
                random_list = {
                    15 = {}
                    1 = {
                        root = {
                            trigger_event = {
                                id = hostile_diplo_event.9
                            }
                        }
                    }
                }
            }
        }

        should_invalidate = {
            OR = {
                scope:target_country.techs_researched <= root.techs_researched
                has_war_with = scope:target_country
                is_country_type = decentralized
                scope:target_country = { is_country_type = decentralized }
                
                #INVALIDATE IF TREATY OF MUTUAL NON INTERFERENCE EXISTS
				scope:target_country = {
					any_scope_treaty = {
						binds = root
						binds = scope:target_country
						any_scope_article = {
							has_type = mutual_non_interference
						}
					}
				}

                in_default = yes
                NOT = {
                    scope:target_country = {
                        any_technology = {
                            is_researched = yes
                            NOT = { root.has_technology_researched = this }
                        }
                    }
                }
                scope:target_country.techs_researched > root.techs_researched
            }
        }


    }

    ai = {
        max_influence_spending_fraction = 0.15
    
        propose_score = {
            value = 100
        }
    
        will_propose = {
            scaled_debt <= 0
            scope:target_country = {
				has_diplomatic_pact = {
					who = root
					type = rivalry
				}
				relations:root <= relations_threshold:cold
			}
        }
        
        will_break = { 
            OR = {        
                scaled_debt > 0.25
            }
        }            
    }

}
