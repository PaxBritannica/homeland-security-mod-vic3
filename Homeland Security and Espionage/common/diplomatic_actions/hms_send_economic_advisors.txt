send_economic_advisors = {
    texture = "gfx/interface/icons/lens_toolbar_icons/send_economic_advisors.dds"

	groups= {
		general
	}

	requires_approval = yes
	should_notify_third_parties = yes

	selectable = {
		NOT = { is_country_type = decentralized }
		is_revolutionary = no
	}

	potential = {
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			is_revolutionary = no
		}
	}

    possible = {

        # TARGET IS NOT MORE THAN A MINOR POWER AND IS MORE THAN COLD TO US
        scope:target_country = {
            any_scope_state = {
                count > 0
            }
            country_rank <= rank_value:minor_power
            relations:root >= relations_threshold:cold
        }

        # INVESTMENT LEVEL NEEDED IN FOREIGN AFFAIRS
        ROOT = {
            # country_rank >= rank_value:major_power
            has_law_or_variant = law_type:law_office_of_foreign_affairs
            institution_investment_level = {
                institution = institution_foreign_affairs
                value >= 2
            }
        }

		#HAS DIPLOMATIC INTEREST
		has_diplomatic_relevance = scope:target_country

        #NOT WAR
        NOT = {
            has_war_with = scope:target_country
        }

		custom_tooltip = {
			text = is_not_building_a_country_already_OR_country_target_is_not_already_being_built_tt

			NOR = {

				#IF TARGET COUNTRY HAS ALREADY AN ECONOMIC ADVISORS IN ACT
				scope:target_country = {
					any_scope_diplomatic_pact = {
						is_diplomatic_action_type = send_economic_advisors
					}
				}
	
				root = {
					any_scope_diplomatic_pact = {
						is_diplomatic_action_type = send_economic_advisors
					}
				}
				
			}
		}
    }

	pact = {
        cost = 250
		is_two_sided_pact = yes
		show_in_outliner = yes

		relations_progress_per_day = 1.6
		relations_improvement_max = 70

        # SOFTER MORE ADVISORISH VERSION THAT AIMS TO NOT CREATE DISCONTENT AMONG THE TARGETS POPS
        # SINCE THE INCOME TRANSFER IS SPREAD MORE EQUALLY
        income_transfer = 0.1
        max_paying_country_income_to_transfer = 0.5
        second_country_gets_income_transfer = yes
        income_transfer_based_on_second_country = yes

        income_transfer_to_pops = {
            allow_non_fully_accepted = yes
            upper_strata_pops = { value = 0.2 }
            middle_strata_pops = { value = 0.3 }
            lower_strata_pops = { value = 0.5 }
        }

        first_modifier = {
            country_pact_leverage_generation_mult = 0.10
        }

        second_modifier = {

            country_construction_goods_cost_mult = -0.2
            state_tax_capacity_mult = 0.2
            state_construction_mult = 0.2

        }

	 	requirement_to_maintain = {
			trigger = {
                has_diplomatic_relevance = scope:target_country
				custom_tooltip = {
					text = DIPLO_ACTION_RELATIONS_POOR_OR_LESS_NOT
					scope:target_country.relations:root > relations_threshold:poor
				}
			}

			show_about_to_break_warning = {
				scope:target_country.relations:root <= -17 # this is: poor + 3
				scope:target_country.relations_change_rate:root < 0
			}
		}

		monthly_effect = {
			custom_tooltip = {
				text = economic_advisors_pact_tt
				if = {
					limit = {
						AND = {
							scope:target_country = {
								relations:root >= 10
							}
							NOT = { has_variable = has_ban_je }
						}
					}
					random_list = { 
						16 = {} # balance ? 
						1 = { 
							root = {
								trigger_event = {
									id = develop_nation_events.1 # Entry event to MAIN JE BAN / BUILD A NATION
								}
							}
						}
					}
				}
				
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_NOT_DECENTRALIZED"
					NOT = { is_country_type = decentralized }
					scope:target_country = {
						NOT = { is_country_type = decentralized }
					}
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "DIPLOMATIC_PACT_REQUIREMENT_AT_PEACE"
					NOT = { has_war_with = scope:target_country }
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = "EQUAL_OR_BETTER_RANK_THAN_TARGET"
					country_rank >= scope:target_country.country_rank
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				in_default = no
			}
		}
	}

	ai = {
		max_influence_spending_fraction = 0.20
	
		evaluation_chance = {
			value = 10
	
			add = {
				value = country_rank
				multiply = 0.05
			}
		}
	
		#Controls AI acceptance
		accept_score = {
			value = 15
			
			# Base acceptance for friendly relations
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = genial
					}
				}
				add = 50
			}
			
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = amicable
					}
				}
				add = 25
			}
			
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = cordial
					}
				}
				add = 15
			}
			
			# Better relations = more likely to accept
			if = {
				limit = {
					relations:scope:actor >= relations_threshold:friendly
				}
				add = 30
			}
			
			if = {
				limit = {
					relations:scope:actor >= relations_threshold:amicable
				}
				add = 15
			}
			
			# If they're technologically behind, they want this more
			if = {
				limit = {
					scope:actor = {
						country_rank < root.country_rank
					}
				}
				add = 30
			}
			
			# Similar strategic interests
			if = {
				limit = {
					has_strategic_adjacency = scope:actor
				}
				add = 20
			}
			
			# # Same power bloc
			# if = {
			# 	limit = {
			# 		AND = {
			# 			has_power_bloc = yes
			# 			scope:actor = {
			# 				has_power_bloc = yes
			# 			}
			# 			power_bloc = scope:actor.power_bloc
			# 		}
			# 	}
			# 	add = 50
			# }
			
			# They owe us an obligation
			if = {
				limit = {
					owes_obligation_to = scope:actor
				}
				add = 100
			}
			
			# Subject relationship
			if = {
				limit = {
					is_subject_of = scope:actor
				}
				add = 200
			}
			
			# Negative modifiers
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = antagonistic
					}
				}
				add = -100
			}
			
			if = {
				limit = {
					has_attitude = {
						who = scope:actor
						attitude = domineering
					}
				}
				add = -50
			}
			
			# They're at war - less interested in science
			if = {
				limit = {
					is_at_war = yes
				}
				add = -20
			}
		}
	
		will_propose = {
			scaled_debt <= 0
			is_at_war = no
			net_fixed_income > 0
			has_attitude = {
				who = scope:target_country
				attitude = protective
			}
		}
	
		will_break = {
			
			has_attitude = {
				who = scope:target_country
				attitude = antagonistic
			}
			
			NOR = {
				root = {
					owes_obligation_to = scope:target_country
				}
				NOT = {
					scope:target_country = {
						owes_obligation_to = root
					}
				}
				AND = {
					is_at_war = no
					scope:target_country = { is_at_war = yes }
				}
			}
		}
	}
}
