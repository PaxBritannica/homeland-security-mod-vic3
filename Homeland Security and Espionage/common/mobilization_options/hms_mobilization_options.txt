mobilization_option_homeletters = {
	texture = "gfx/interface/icons/pms_icons/homeletters.dds"

	unlocking_technologies = {
		military_statistics
	}

	possible = {
		market ?= {
			mg:intel ?= {
				has_potential_supply = yes
			}
			mg:paper ?= {
				has_potential_supply = yes
			}
		}
	}

    on_deactivate_while_mobilized = {
		custom_tooltip = {
			text = mobilization_option_it_hurts_morale_when_you_remove_supplies_while_in_combat_tt
			every_combat_unit = {
				add_morale = {
					value = morale
					multiply = -0.3
				}
			}
		}
	}

	upkeep_modifier = {
		goods_input_intel_add = 1
		goods_input_paper_add = 1
	}

	unit_modifier = {
		unit_morale_loss_mult = -0.05
        unit_morale_recovery_mult = 0.1
	}

	ai_weight = {
		value = 1
	}

	group = supplies
}
