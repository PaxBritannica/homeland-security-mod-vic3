#RAILROADS BAR 
national_railway_project_bar = {
	name = "national_railway_project_bar"
	desc = "national_railway_project_bar_desc"

    default = yes

    ### Build a Nation - Railroad Progress Bar Monthly Growth ###
### All variables influence the bar's growth rate ###

monthly_progress = {

    
    # BASE GROWTH
    if = {
        limit = {
            has_variable = ban_railroad_construction_active
        }
        add = {
            desc = "ban_base_growth_tt"
            value = 2
        }
    }

    # Minimum wage workers cheap labor, small boost
    if = {
        limit = {
            has_variable = var_min_wage_workers
        }
        add = {
            desc = "ban_min_wage_workers_tt"
            value = 0.5
        }
    }

    # Extensive hiring lots of workers, solid boost
    if = {
        limit = {
            has_variable = var_LOT_paid_wage_workers
        }
        add = {
            desc = "ban_LOT_paid_wage_workers_tt"
            value = 1
        }
    }

    # Selective professional hiring, best workers, best boost
    if = {
        limit = {
            has_variable = var_PRO_paid_wage_workers
        }
        add = {
            desc = "ban_PRO_paid_wage_workers_tt"
            value = 2
        }
    }

    # =============================================
    # NEGATIVE Worker condition debuffs

    # Workers healing small slowdown while they recover
    if = {
        limit = {
            has_variable = var_healing_workers
        }
        add = {
            desc = "ban_healing_workers_tt"
            value = -1
        }
    }

    # Workers resting at slower pace ,slight delay
    if = {
        limit = {
            has_variable = var_workers_are_working_slower_pace
        }
        add = {
            desc = "ban_workers_slower_pace_tt"
            value = -1
        }
    }

    # Workers are overworked, noticeable slowdown
    if = {
        limit = {
            has_variable = var_workers_are_overworked
        }
        add = {
            desc = "ban_workers_overworked_tt"
            value = -1
        }
    }

    # Refused to cure sick workers, morale collapsed
    if = {
        limit = {
            has_variable = var_refused_curing_workers
        }
        add = {
            desc = "ban_refused_curing_tt"
            value = -2
        }
    }
}
    

	start_value = 0
	min_value = 0
	max_value = 50
}
