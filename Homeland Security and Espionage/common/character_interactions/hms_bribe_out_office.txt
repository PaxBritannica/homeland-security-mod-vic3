bribe_out_of_office = {
    # Actors who are not either Monarchs or Theocratic rulers can be bribed out of office.

    show_requirements = yes

    potential = {
		is_in_exile_pool = no
		is_in_void = no

		OR = {
			is_heir = yes
			has_role = general
			has_role = admiral
			has_role = politician
			has_role = agitator
			has_role = executive
		}

        has_technology_researched = HMS_tech_era_2_secret_service

		is_immortal = no
	}

    

    possible = {}

    effect = {
        this = {
            save_scope_as = resigning_politician
        }

        owner ?= {
            trigger_event = { id = hms_resignation_events.1 popup = yes }
        }
	}

    cooldown = { months = 24 }
    show_confirmation_box = yes

    ai_chance = {
        value = 0

        # AI will consider if agitator is causing major problems
        if = {
            limit = {
                popularity >= 75
                interest_group ?= {
                    is_marginal = no
                    ig_approval >= happy
                }
            }
            add = 50
        }

        if = {
            limit = {
                popularity <= 0
                interest_group ?= {
                    is_marginal = no
                    ig_approval <= unhappy
                }
            }
            add = 50
        }

        # Giving too many debug errors for testing other features,
        # activate once we can focus here
        # #Avoid if agitator supports laws AI wants
        # if = {
        #     limit = {
        #         scope:actor = {
        #             any_political_movement = {
        #                 exists = most_desired_law
        #                 leader = root
        #             }
        #         }
        #     }
        #     add = -100
        # }
    }
}
