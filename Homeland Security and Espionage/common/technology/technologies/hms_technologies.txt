# ERA 2
HMS_tech_era_2_secret_service = {

	#Unlocks the Headquarter Building

	era = era_2
	category = society
	texture = "gfx/interface/icons/invention_icons/secret_service_wbg.dds"
	unlocking_technologies = { nationalism, central_archives }
	

	# Bonuses provided mainly for flavor and to boost AI interest
	modifier = {
		country_prestige_mult_ = 0.05
		country_influence_mult = 0.1
		country_authority_mult = 0.1
		country_agitator_slots_add = 1

		state_migration_pull_mult = -0.1

		# For future we will have our own institution and so we will add institutional max level add
		# eg as seen for workplace safety tech investment
		# country_institution_workplace_safety_max_investment_add = 3
	}

	# triggers event
	on_researched = {

		set_global_variable = hms_tech_secret_service_researched

		trigger_event = {
			id = hms_secret_service_events.1
			days = 0
		}
	}

	#ai research propensity
	ai_weight = {
		value = 2

		if = {
			limit = {
				OR = {
					has_law = law_type:law_autocracy
					has_law = law_type:law_oligarchy
				}
			}
			multiply = 1.5
		}

		# Great powers prioritize it
        if = {
            limit = { is_great_power = yes }
            multiply = 1.3
        }
        
        # Higher priority if in rivalries
        if = {
            limit = {
                any_scope_country = {
                    has_diplomatic_pact = {
                        who = ROOT
                        type = rivalry
                    }
                }
            }
            multiply = 1.4
        }
	}

}

HMS_tech_era_3_extended_networks = {

	era = era_3
	category = society
	unlocking_technologies = { HMS_tech_era_2_secret_service }
	texture = "gfx/interface/icons/invention_icons/scientific_division.dds"

	# Bonuses provided mainly for flavor and to boost AI interest
	modifier = {

		country_prestige_mult = 0.05
		country_influence_mult = 0.1

		country_society_tech_spread_mult = 0.1
		country_society_tech_research_speed_mult = 0.05

		country_institution_foreign_affairs_max_investment_add = 2
	}
	
	#ai research propensity
	ai_weight = {
		value = 2

		if = {
			limit = {
				OR = {
					has_law = law_type:law_autocracy
					has_law = law_type:law_oligarchy
				}
			}
			multiply = 1.5
		}

		# Great powers prioritize it
        if = {
            limit = { is_great_power = yes }
            multiply = 1.3
        }
        
        # Higher priority if in rivalries
        if = {
            limit = {
                any_scope_country = {
                    has_diplomatic_pact = {
                        who = ROOT
                        type = rivalry
                    }

                }
            }
            multiply = 1.3
        }

		if = {
			limit = {
				ruler = {
					has_ideology = ideology_traditionalist
				}
			}
			value = 0.8
		}

		if = {
			limit = {
				has_strategy = ai_strategy_progressive_agenda
			}
			multiply = 1.1
		}
	}
}
