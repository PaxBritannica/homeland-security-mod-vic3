INJECT_OR_CREATE:building_military_intel_hq = {
	building_group = bg_intel_buildings
	city_type = city
	ownership_type = self

	has_max_level = yes

	icon = "gfx/interface/icons/building_icons/military_intel_hq.dds"
	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"

	# has_max_level = yes needs variable to be implemented with relative modifier
	# levels_per_mesh = 50
	
	unlocking_technologies = { military_statistics }
	required_construction = construction_cost_high
	

	production_method_groups = {
		hms_pmg_group_2
	}

	ai_value = {
		value = 800

		# Great powers should always want military intel
		if = {
			limit = { 
				owner ?= {
					country_rank >= rank_value:great_power
				} 
			}
			multiply = 1.5
		}

		# Major powers too
		if = {
			limit = { 
				owner ?= {
					country_rank >= rank_value:major_power
				} 
			}
			multiply = 1.1
		}

		if = {
			limit = { 
				owner ?= {
					country_rank >= rank_value:minor_power
				} 
			}
			multiply = 0.03
		}
	}
}
