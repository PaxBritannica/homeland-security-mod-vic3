je_sabotage_enforce_martial_law = {
	name = hms_enforce_martial_law
	desc = hms_enforce_martial_law_desc

	visible = { always = yes }
	
	ai_chance = {
		base = 5
	}

	possible = {
		authority >= 200
		# NOT = { is_at_war = yes } Technically Martial Law can still be enacted during wartime
		army_size >= 20

		NOT = {
			root = { has_law_or_variant = law_type:law_militarized_police }
		}
	}

	effect = {
		root = {
			activate_law = law_type:law_militarized_police
			country_authority_add = -200
		}

		# add_progress = {
		# 	name = sabotage_police_progress_bar
		# 	value = 10
		# }

		set_variable = {
			name = enacted_martial_law
			months = 12
		}
	}
}
