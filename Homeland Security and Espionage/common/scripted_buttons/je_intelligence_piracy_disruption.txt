#Fight them with navy
hms_je_piracy_fought = {
	name = hms_je_piracy_fought
	desc = hms_je_piracy_fought_desc

	visible = { always = yes }
	
	ai_chance = {
		base = 5
	}

	possible = {
		AND = {
			navy_size >= 20
			not = {
				has_variable = recently_fought_pirates
			}
		}
	}

	effect = {
		trigger_event = {id = hostile_diplo_events.6} # Engage the pirates

		root = {
			set_variable = {
				name = recently_fought_pirates
				months = 3
			}
			
			post_notification = stopped_attack_message
		}
		
	}
}
